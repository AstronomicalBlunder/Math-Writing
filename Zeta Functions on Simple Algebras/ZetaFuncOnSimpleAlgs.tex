\section{Convergence Lemmas}
In this section, we establish the convergence of some integrals which will be needed later. Let $F$ be a (commutative) local field and $H$ a division algebra with center $F$. Let $M$ be the simple algebra of $m \times m$ matrices with entries in $H$. If $d^2$ is the rank of $H$ over $F$, then rank of $M$ over $F$ is $n^2$, where $n = md$.

\begin{example}[Prototypical Example]
The simplest non-trivial example of the above setup would be when $H = F$, and 
\begin{equation*}
    M = \mat_{n \times n}(F), \ M^\times = \GL_n(F).
\end{equation*}
\end{example}

We recall some terminology and properties about central simple algebras.

\begin{defn}
\label{def: CSA splitting field}
Let $F$ be a field, and $A$ a central simple $F$-algebra (i.e $A$ is a finite-dimensional associative $F$-algebra whose center is $F$). A finite extension $K/F$ is said to be a \textbf{splitting field for} $A$ if 
\begin{equation*}
    K \tensor_F A \cong \mat_{n \times n}(K)
\end{equation*}
for some $n$. We take it as a fact here (which is not too hard to show) that every such $A$ admits a splitting field.
\end{defn}

As a consequence, if we let $K$ be the splitting field of $H$ over $F$ and $G$ the unit group of $M$, the above definition/fact tells us that any such $G$ is of the form $\mat_{m \times m}^\times(H)$, and thus base changing to $K$ gives:
\begin{equation*}
    G_K \cong \mat_{m \times m}^\times(K \tensor_F H) \cong \GL_{m}(\mat_{k \times k}(K)). 
\end{equation*}
Thus we deduce that any such $G$ is a \textbf{reductive} algebraic group. 

\begin{defn}
\label{def: reduced norm and trace}
With the setup as above, the \textbf{reduced norm}, denoted $\nu_M: M \to F$, is the unique polynomial maps such that the following diagram commutes:

\begin{center}
    \begin{tikzcd}
    M \arrow[r, " \nu_M"] \arrow[d, hookrightarrow] & F \arrow[d, hookrightarrow]\\
    K \tensor M \arrow[r, "\nu_M \tensor \bone"] \arrow[d, "\sim" {rotate=90, anchor=north}]  & K \arrow[d, equals] \\
    \mat_{k \times k}(K) \arrow[r, "\det"] & K.
    \end{tikzcd}\\
\end{center}

The \textbf{reduced trace} $\tau_M : M \to F$ is defined similarly.

\end{defn}

\section{Induced Representations}

Throughout this section, $F$ is a non-Archimedean local field, and $M$ is a semi-simple algebra over $F$ such that
\begin{equation*}
  M \cong \bigdsum_i M_i,
\end{equation*}
where each of the $M_i$ are central simple $F$-algebras.

The multiplicative group $G$ of such an $M$ as above is a reductive algebraic group defined over $F$ (whose $R$-rational points we denote $G(R)$ for any $F$-algebra $R$, though we often just write $G$ if context is clear) and clearly we have
\begin{equation}
  G \cong \prod_i G_i,
\end{equation}
where $G_i := M_i^\times$. 

We are interested in representations $\pi$ of $G(F)$ -- henceforth denoted as just $G$ -- on a $\cc$-vector space $V$ that are \textit{smooth and admissible}.
\begin{defn}
  \label{def: smooth and admissible repns}
  A representation $(\pi, V)$ of $G$ is said to be \textbf{smooth} if $\stab_G(v)$ is an open subgroup of $G$ for every $v \in V$. It is \textbf{admissible} if it is smooth and and $V^{H}$ (the space of vectors in $V$ stabilized by $H$) is finite dimensional for every compact open subgroup $H$.
\end{defn}

\begin{remark}
One can verify that smoothness of a representation is equivalent to continuity of the map
\begin{align*}
G \times V &\to V\\
(g, v) &\mapsto \pi(g) v
\end{align*}
if we give $V$ the trivial locally convex topology in which every seminorm is continuous. 
\end{remark}



\begin{defn}
  \label{def: hecke algebra}
We denote by $\fanh(G(F))$ the space of locally constant and compactly supported complex-valued functions on $G(F)$. Convolution turns this into a complex algebra, known as the \textbf{Hecke algebra}. 
\end{defn}


Suppose $(\pi, V)$ is a smooth representation of $G$. Then for any $f \in \fanh(G)$ we may define an operator $\pi(f)$ on $V$ in the following manner:
\begin{equation}
  \label{eqn: repn of H(G)}
  \pi(f) : v \mapsto \int_G f(g) \pi(g) v \dg.
\end{equation}
Note by assumption $f$ is locally constant and compactly supported, so the above integral is in fact a finite linear combination of vectors in $V$. Thus we may also view $\pi : f \mapsto \pi(f)$ as a representation of $\fanh(G)$ on $V$.

If we denote by $\lambda$ the shift operator on $\fanh(G)$, i.e 
\begin{equation*}
  \lambda(g)f (h) := f( \inv{g}h),
\end{equation*}
then we note we have the identity
\begin{align*}
  \pi (\lambda(g)f) v &= \int_G f(\inv{g}h) \pi(h)v \dh\\
  &= \int_G f(h) \pi(hg) v \dh\\
  &= \int_G f(h) \pi(g) \pi(h) v \dh\\
  &= \pi(g)\pi(f) v.
\end{align*}

\begin{lemma}
  \label{lemma: admissible repns}
A representation $(\pi, V)$ of $G$ is smooth if and only if the induced representation $\pi$ on $\fanh(G)$ satisfies that for every $v \in V$ there exists some $f \in \fanh(G)$ so that $\pi(f)v = v$.
\end{lemma}
\begin{proof}
If $\pi$ is smooth, then by assumption $\stab_G(v)$ is an open (hence closed!) subgroup. We may pick $f$ then to be an appropriate multiple of the characteristic function of a  compact open set $C$ within this open subgroup (specifically it should be multiplied by the reciprocal of the Haar measur of the compact open), which would immediately give
\begin{equation*}
  \pi(f)v := \int_G f(g) \pi(g) v \dg = \frac{1}{\vol(C)} \int_C \pi(g)v \dg = \frac{1}{\vol(C)} \cdot \vol(C) v = v.
\end{equation*}
Conversely, suppose we have a representation $(\pi, V)$ of $\fanh(G)$ so that every vector $v$ has some $f$ so that $\pi(f)v = v$. We will construct a smooth $(\pi', V)$ representation of $G$ so that
\begin{equation*}
  \pi(f) v = \int_G f(g) \pi'(g) v \dg
\end{equation*}
for every $v \in V$.


\end{proof}

\section{Automorphic Forms}
In this chapter $K$ will be a global field, $H$ will be a division algebra of center $F$ and rank $d^2$, $M$ will be the simple algebra $\mat_{m \times m}(H)$, and $G$ will be group $M^\times$. We will denote by $\aa_K$ and $\ii_K$ the adeles and ideles of $K$ respectively. For each place $v \in \fanp_K$ we denote by $K_v$ the (local!) completion of $K$ at $v$ and put
\begin{equation*}
  H_v := H \tensor_K K_v, \quad M_v := M \tensor_K K_v, \quad G_v := \GL_m(H_v).
\end{equation*}
For each $v$ we choose an isomorphism
\begin{equation}
  \label{eqn: H_v isom}
  H_v \bij \mat_{m_v' \times m_v'}(H_v'),
\end{equation}
where $H_v'$ is a division algebra of center $K_v$ and rank $(d_v')^2$, and $d = m_v' d_v'$. From \ref{eqn: H_v isom}, we derive isomorphisms
\begin{equation}
  \label{eqn: M_v, G_v isoms}
  M_v \bij \mat_{m m_v' \times mm_v'}(H_v'), \quad G_v \bij \GL_{m m_v'}(H_v').
\end{equation}
The isomorphism in (\ref{eqn: H_v isom}) -- and therefore the isomorphisms in \ref{eqn: M_v, G_v isoms} -- are defined up to an inner automorphism. In particular, our choice of isomorphism does not affect the representation theory.

For each $v \in \fanp_K$ we also define the constant
\begin{equation}
  \label{eqn: a_v constant}
a_v := (-1)^{m m_v' (d_v' - 1)}
\end{equation}
Note that for almost all (i.e. cofinitely many) $v$ we have
\begin{equation*}
  H_v = \mat_{d_v \times d_v}(K_v), \ H_v' = K_v, \ M_v = \mat_{n \times n}(K_v), \ G_v = \GL_n(K_v),
\end{equation*}
with
\begin{equation*}
  d_v' = 1, \ m_v' = 1, \ a_v = 1.
\end{equation*} 
It is an easy consequence of the reciprocity law (in the Hasse form) that
\begin{equation*}
  \prod_{v \in \fanp_K} a_v = 1.
\end{equation*}
We shall give a direct proof of this later {\color{red} [Where?]}.

\end{equation*}  
where $R_v'$ is the ring of integers in $H_v'$.

The product

\end{equation*}
We denote by $G_0$ the subgroup of $g \in G(\aa_K)$ such that $\abs{\nu_M(g)} = 1$.

\begin{defn}
\label{def: cuspidal func on G_A/G_K}
A continuous function $\phi$ on $G(K) \backslash G(\aa_K)$ is said to be \textbf{cuspidal} if the integral
\begin{equation*}
  \int_{U_K \backslash U(\aa_K)} \phi(u g) \du
\end{equation*}

\end{defn}

Let $T$ be a maximal $K$-split torus of $G$ and $\alpha_i$ (for $i \in [1, m-1]$) a system of simple roots of $T$. We put
\begin{equation*}
  \eta(t) := \inf_{i} \abs{\alpha_i(t)}, \quad \kappa(t) := \sup_i \abs{\alpha_i(t)}.
\end{equation*}

\begin{defn}
\label{def: slowly increasing + rapidly decreasing function on G_A/G_K}
A function $\phi$ on $G(K) \backslash G(\aa_K)$ is said to be \textbf{slowly increasing} (resp. \textbf{rapidly decreasing}) if for any compact $C \subseteq G$ and any $c > 0$ we have $p \geq 1$ and $c' > 0$ such that the relations
\begin{equation*}
  g \in C, \ t \in T(\aa_K) \cap G_0, \ \eta(t) \geq c
\end{equation*}
imply
\begin{equation*}
  \abs{\phi(tg)} \leq c' \kappa(t)^{p} \text{ (resp. }  \abs{\phi(tg)} \leq c' \kappa(t)^{-p} \text{)}
\end{equation*}
